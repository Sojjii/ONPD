\usepackage{caption}
\DeclareCaptionLabelSeparator{emdash}{ — }  % Створюємо розділювач "довге тире з пробілами"

\captionsetup[figure]{
	labelsep=emdash,         % Використовувати довге тире
	name={Рисунок},          % Замість стандартного "Рис." пишемо повністю "Рисунок"
	justification=centering, % Центрувати підпис під рисунком
	position=bottom,         % Позиція за замовчуванням — знизу
	font=small               % За ДСТУ підписи часто роблять трохи меншим шрифтом (наприклад, 12pt, якщо текст 14pt)
}

\captionsetup[table]{
	labelsep=emdash,
	name={Таблиця},
	justification=raggedright, % За ДСТУ підпис таблиці зазвичай вирівнюють по лівому краю
	singlelinecheck=false,     % Навіть короткий підпис залишається ліворуч
	position=top               % Позиція за замовчуванням — зверху
}

	\counterwithin{table}{chapter}
}
